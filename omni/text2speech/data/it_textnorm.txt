# Italian text normalization for the Kokoro TTS front end.
#
# Format is documented in omni/tts/text_normalizer.h. Records are
# tab-separated, and `rule` records apply in the order they appear here.
#
# The phonemizer hands whole clauses to espeak-ng. This table rewrites what
# espeak-ng reads poorly or not at all: locale number separators (1.500,50),
# currency, temperatures, clock times, slash dates, common fractions,
# ordinals and a few abbreviations.
#
# It then spells out every remaining number, ordinal and Roman numeral in
# words, so espeak-ng's own number reader never runs for Italian: that reader
# (LookupNum2 in the bundled espeak-ng) indexes before the start of a buffer
# for Italian single digits and whole-word numbers. Numbers are split into
# the same words espeak-ng would speak (2026 -> due mila ventisei,
# 1500 -> mille cinquecento, 108 -> cento otto) to keep Host parity.

# `#N` reads digits one at a time; each digit becomes a separate word.
digit	0	 zero
digit	1	 uno
digit	2	 due
digit	3	 tre
digit	4	 quattro
digit	5	 cinque
digit	6	 sei
digit	7	 sette
digit	8	 otto
digit	9	 nove

# Phone numbers are read digit by digit: +39 02 1234567.
rule	\+(\d{1,3}) (\d{2,4}) (\d{5,8})\b	più #1 #2 #3

# Abbreviations.
rule	\bdott\.	dottor
rule	\bDott\.	Dottor
rule	\bsig\.	signor
rule	\bS\.p\.A\.	esse pi a

# Temperatures.
rule	-\s?(\d+(?:,\d+)?)\s?°\s?C\b	meno \1 gradi Celsius
rule	(\d)\s?°\s?C\b	\1 gradi Celsius

# Ordinals (1º, 21º) and Roman numerals (XXI secolo, Papa Giovanni XXIII)
# are marked ⟪N⟫ here and become ordinal words at the end of the table.
# espeak-ng reads all-caps Roman numerals from II to XLIX, and V./X. with a
# dot, as ordinals.
rule	\b(\d{1,3})[ºª]	⟪\1⟫
rule	\b(10|[1-9])°	⟪\1⟫
rule	\b(XLIX|XLVIII|XLVII|XLVI|XLV|XLIV|XLIII|XLII|XLI|XL|XXXIX|XXXVIII|XXXVII|XXXVI|XXXV|XXXIV|XXXIII|XXXII|XXXI|XXX|XXIX|XXVIII|XXVII|XXVI|XXV|XXIV|XXIII|XXII|XXI|XX|XIX|XVIII|XVII|XVI|XV|XIV|XIII|XII|XI|IX|VIII|VII|VI|IV|III|II)\b	⟪{roman:\1}⟫
rule	\b([VX])\.	⟪{roman:\1}⟫.

# Clock times: 10:15 -> dieci e un quarto, 14:30 -> quattordici e trenta.
rule	\b(\d{1,2}):15\b	\1 e un quarto
rule	\b(\d{1,2}):00\b	\1
rule	\b(\d{1,2}):([0-5]\d)\b	\1 e \2

# Slash dates (day/month/year) and common fractions.
rule	\b(\d{1,2})/(\d{1,2})/(\d{4})\b	\1 {month:\2} \3
rule	\b(\d/\d)\b	{fraction:\1}

# Thousands separators.
rule	(\d)\.(\d\d\d)\b	\1\2
rule	(\d)\.(\d\d\d)\b	\1\2
rule	(\d)\.(\d\d\d)\b	\1\2

# Currency: the symbol precedes the amount but the unit follows it.
rule	€\s?(\d+),(\d\d)\b	\1 euro e \2 centesimi
rule	(\d+),(\d\d)\s?€	\1 euro e \2 centesimi
rule	€\s?(\d+)	\1 euro
rule	(\d+)\s?€	\1 euro
rule	\$\s?(\d+),(\d\d)\b	\1 dollari e \2 centesimi
rule	\$\s?(\d+)	\1 dollari

# Temperatures without a unit.
rule	(\d)\s?°	\1 gradi

# Decimal comma and percent.
rule	(\d),(\d)	\1 virgola \2
rule	(\d)\s?%	\1 per cento

# A minus sign before a number that does not follow a digit or letter.
rule	(^|[^\w])-(\d)	\1meno \2

# Spell out every remaining number. Keep the words apart from adjacent letters
# (A4 -> A quattro).
rule	(\pL)(\d)	\1 \2
rule	(\d)(\pL)	\1 \2
rule	(\d+)	⟦\1⟧
# A number with a leading zero (a code such as 007, or the minutes of 14:05)
# is read one digit at a time.
rule	⟦(0\d+)⟧	#1
# More than 12 digits: one digit at a time.
rule	⟦(\d{13,})⟧	#1
# Split into miliardi / milioni / migliaia groups of three digits.
rule	⟦(\d{1,3})(\d{9})⟧	⟦B\1⟧⟦\2⟧
rule	⟦(\d{1,3})(\d{6})⟧	⟦M\1⟧⟦\2⟧
rule	⟦(\d{1,3})(\d{3})⟧	⟦K\1⟧⟦\2⟧
rule	⟦[BMK]0+⟧	
rule	⟧⟦0+⟧	⟧
rule	⟧⟦	⟧ ⟦
rule	⟦K0*1⟧	mille
rule	⟦M0*1⟧	un milione
rule	⟦B0*1⟧	un miliardo
rule	⟦K(\d+)⟧	⟦\1⟧ mila
rule	⟦M(\d+)⟧	⟦\1⟧ milioni
rule	⟦B(\d+)⟧	⟦\1⟧ miliardi
# Hundreds: 108 -> cento otto, 500 -> cinquecento.
rule	⟦0(\d\d)⟧	{n99:\1}
rule	⟦(\d)00⟧	{h:\1}
rule	⟦(\d)0(\d)⟧	{h:\1} {n99:\2}
rule	⟦(\d)(\d\d)⟧	{h:\1} {n99:\2}
rule	⟦(\d\d?)⟧	{n99:\1}

# Ordinal words: primo..decimo, then drop the final vowel and add -esimo
# (ventuno -> ventunesimo, cento -> centesimo, ventitré -> ventitreesimo).
rule	⟪\s*(uno|due|tre|quattro|cinque|sei|sette|otto|nove|dieci)⟫	{ordsmall:\1}
rule	⟪\s*([^⟫]*)tré⟫	\1treesimo
rule	⟪\s*([^⟫]*)sei⟫	\1seiesimo
rule	⟪\s*([^⟫]*)[aeiou]⟫	\1esimo
rule	⟪\s*([^⟫]*)⟫	\1
