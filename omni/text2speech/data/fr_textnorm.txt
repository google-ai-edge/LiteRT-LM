# French text normalization for the Kokoro TTS front end.
#
# Format is documented in omni/tts/text_normalizer.h. Records are
# tab-separated, and `rule` records apply in the order they appear here.
#
# The phonemizer hands whole clauses to espeak-ng, which reads plain cardinals
# and ordinals (1er, 2e) itself. This table rewrites what espeak-ng reads
# poorly or not at all: locale number separators (15 000, 9,99), currency,
# temperatures, clock times, slash dates and common fractions.

# Ordinals: 2e / 2ème -> deuxième.
rule	\b([2-9]|1\d|20)(?:e|ème)\b	{ordinal:\1}

# Clock times: 10h15 / 10:15 -> 10 heures 15.
rule	\b(\d{1,2})[h:]00\b	\1 heures
rule	\b(\d{1,2})[h:]([0-5]\d)\b	\1 heures \2

# Slash dates (day/month/year) and common fractions.
rule	\b(\d{1,2})/(\d{1,2})/(\d{4})\b	\1 {month:\2} \3
rule	\b(\d/\d)\b	{fraction:\1}

# Thousands separators: a space (or no-break space) or a dot.
rule	(\d)[ \x{00A0}\x{202F}](\d\d\d)\b	\1\2
rule	(\d)[ \x{00A0}\x{202F}](\d\d\d)\b	\1\2
rule	(\d)\.(\d\d\d)\b	\1\2
rule	(\d)\.(\d\d\d)\b	\1\2

# Currency: the symbol precedes the amount but the unit follows it.
rule	(\d+),(\d\d)\s?€	\1 euros et \2 centimes
rule	€\s?(\d+),(\d\d)\b	\1 euros et \2 centimes
rule	(\d+)\s?€	\1 euros
rule	€\s?(\d+)	\1 euros
rule	(\d+),(\d\d)\s?CHF\b	\1 francs suisses et \2 centimes
rule	(\d+)\s?CHF\b	\1 francs suisses

# Temperatures.
rule	-\s?(\d+(?:,\d+)?)\s?°\s?C\b	moins \1 degrés Celsius
rule	(\d)\s?°\s?C\b	\1 degrés Celsius
rule	(\d)\s?°	\1 degrés

# Decimal comma and percent.
rule	(\d),(\d)	\1 virgule \2
rule	(\d)\s?%	\1 pour cent
