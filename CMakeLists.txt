# LiteRT-LM Orchestrator
cmake_minimum_required(VERSION 3.25 FATAL_ERROR)
project(LiteRT-LM-Orchestrator LANGUAGES NONE)

include(ExternalProject)

option(ENABLE_CCACHE "Enables the use of ccache" OFF)
if(${ENABLE_CCACHE} STREQUAL "ON")
  find_program(CCACHE_PROGRAM ccache)
  if(CCACHE_PROGRAM)
      message(STATUS "Using CCache to speed up builds")
      set(CMAKE_C_COMPILER_LAUNCHER "${CCACHE_PROGRAM}")
      set(CMAKE_CXX_COMPILER_LAUNCHER "${CCACHE_PROGRAM}")
  else()
      message(WARNING "CCache not found. Install it (`sudo apt install ccache`) for faster builds!")
  endif()
endif()

set(LITERTLM_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "LiteRT-LM: Absolute path to the repository root")
set(LITERTLM_TOOLCHAIN_ARGS "" CACHE STRING "String used to define parameterized CMAKE_ARGS")
set(LITERTLM_PREBUILD_DIR "${CMAKE_CURRENT_BINARY_DIR}/prebuild")
set(LITERTLM_PREBUILD_BINARY_DIR "${LITERTLM_PREBUILD_DIR}/build")
set(LITERTLM_PREBUILD_STAMP_DIR "${LITERTLM_PREBUILD_DIR}/stamps")
set(LITERTLM_PREBUILD_TMP_DIR "${LITERTLM_PREBUILD_DIR}/tmp")
set(LITERTLM_PREBUILD_DOWNLOAD_DIR "${LITERTLM_PREBUILD_DIR}/download")
set(LITERTLM_PREBUILD_INSTALL_DIR "${LITERTLM_PREBUILD_DIR}/install")
set(LITERTLM_PREBUILD_EXTERNAL_PROJECT_DIR "${LITERTLM_PREBUILD_BINARY_DIR}/external")
set(LITERTLM_DIR "${CMAKE_CURRENT_BINARY_DIR}/litert_lm")
set(LITERTLM_BINARY_DIR "${LITERTLM_DIR}/build")
set(LITERTLM_STAMP_DIR "${LITERTLM_DIR}/stamps")
set(LITERTLM_TMP_DIR "${LITERTLM_DIR}/tmp")
set(LITERTLM_DOWNLOAD_DIR "${LITERTLM_DIR}/download")
set(LITERTLM_INSTALL_DIR "${LITERTLM_DIR}/install")
set(LITERTLM_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake/packages/litert_lm")
set(LITERTLM_PREBUILD_TARGETS "absl gtest protobuf flatbuffers")
set(LITERTLM_PREBUILD_EXE "protoc flatc")

if(DEFINED LITERTLM_CC AND DEFINED LITERTLM_CXX)
  message(STATUS "[LiteRTLM] Prebuild: Using host compilers: CC=${LITERTLM_CC}, CXX=${LITERTLM_CXX}")
  set(LITERTLM_HOST_CC_COMPILER ${LITERTLM_CC} CACHE PATH "LiteRT-LM: C compiler for prebuild phase")
  set(LITERTLM_HOST_CXX_COMPILER ${LITERTLM_CXX} CACHE PATH "LiteRT-LM: C++ compiler for prebuild phase")
else()
  message(STATUS "[LiteRTLM] Prebuild: Using default host compilers: CC=${CMAKE_C_COMPILER}, CXX=${CMAKE_CXX_COMPILER}")
  set(LITERTLM_HOST_CC_COMPILER ${CMAKE_C_COMPILER} CACHE PATH "LiteRT-LM: C compiler for prebuild phase")
  set(LITERTLM_HOST_CXX_COMPILER ${CMAKE_CXX_COMPILER} CACHE PATH "LiteRT-LM: C++ compiler for prebuild phase")
endif()

if(DEFINED LITERTLM_CC_NDK AND DEFINED LITERTLM_CXX_NDK)
  message(STATUS "[LiteRTLM] Target: Using NDK compilers: CC_NDK=${LITERTLM_CC_NDK}, CXX_NDK=${LITERTLM_CXX_NDK}")
  set(LITERTLM_TARGET_CC_COMPILER ${LITERTLM_CC_NDK} CACHE PATH "LiteRT-LM: C compiler for NDK phase")
  set(LITERTLM_TARGET_CXX_COMPILER ${LITERTLM_CXX_NDK} CACHE PATH "LiteRT-LM: C++ compiler for NDK phase")
else()
  message(STATUS "[LiteRTLM] Target: host compilers: CC=${LITERTLM_HOST_CC_COMPILER}, CXX=${LITERTLM_HOST_CXX_COMPILER}")
  set(LITERTLM_TARGET_CC_COMPILER ${LITERTLM_HOST_CC_COMPILER} CACHE PATH "LiteRT-LM: C compiler for target phase")
  set(LITERTLM_TARGET_CXX_COMPILER ${LITERTLM_HOST_CXX_COMPILER} CACHE PATH "LiteRT-LM: C++ compiler for target phase")
endif()

ExternalProject_Add(
  prebuild
  SOURCE_DIR   "${LITERTLM_PACKAGE_SOURCE_DIR}"
  PREFIX       "${LITERTLM_PREBUILD_DIR}"
  BINARY_DIR   "${LITERTLM_PREBUILD_BINARY_DIR}"
  STAMP_DIR    "${LITERTLM_PREBUILD_STAMP_DIR}"
  TMP_DIR      "${LITERTLM_PREBUILD_TMP_DIR}"
  DOWNLOAD_DIR "${LITERTLM_PREBUILD_DOWNLOAD_DIR}"
  INSTALL_DIR  "${LITERTLM_PREBUILD_INSTALL_DIR}"
  CONFIGURE_COMMAND 
    ${CMAKE_COMMAND} -E env CC=${LITERTLM_HOST_CC_COMPILER} CXX=${LITERTLM_HOST_CXX_COMPILER}
    ${CMAKE_COMMAND} -S ${LITERTLM_PACKAGE_SOURCE_DIR} -B ${LITERTLM_PREBUILD_BINARY_DIR} 
    "-DCMAKE_INSTALL_PREFIX=${LITERTLM_PREBUILD_INSTALL_DIR}"
    "-DLITERTLM_ORCHESTRATION_PHASE=prebuild"
    "-DLITERTLM_PREBUILD_PHASE=LITERT_PREBUILD_PHASE"
    "-DLITERTLM_PROJECT_ROOT=${LITERTLM_PROJECT_ROOT}"
    "-DLITERTLM_PREBUILD_TARGETS=${LITERTLM_PREBUILD_TARGETS}"
    "-DLITERTLM_PREBUILD_EXTERNAL_PROJECT_DIR=${LITERTLM_PREBUILD_EXTERNAL_PROJECT_DIR}"
    "-DLITERTLM_BUILD_FETCH_CONTENT=OFF"
    "-DLITERTLM_BUILD_EXTERNAL_PROJECTS=OFF"
    "-DLITERTLM_BUILD_TARGETS=OFF"
  INSTALL_COMMAND ""
)


# --- Host Tool Setup ---
if(NOT DEFINED LITERTLM_HOST_PROTOC)
    set(LITERTLM_HOST_PROTOC "${LITERTLM_PREBUILD_EXTERNAL_PROJECT_DIR}/protobuf/install/bin/protoc" CACHE PATH "")
    set(LITERTLM_HOST_PROTOC_BIN_DIR "${LITERTLM_PREBUILD_EXTERNAL_PROJECT_DIR}/protobuf/install/bin" CACHE PATH "")
endif()

if(NOT DEFINED LITERTLM_HOST_FLATC)
    set(LITERTLM_HOST_FLATC "${LITERTLM_PREBUILD_EXTERNAL_PROJECT_DIR}/flatbuffers/install/bin/flatc" CACHE PATH "")
    set(LITERTLM_HOST_FLATC_BIN_DIR "${LITERTLM_PREBUILD_EXTERNAL_PROJECT_DIR}/flatbuffers/install/bin" CACHE PATH "")
endif()

if(DEFINED CMAKE_TOOLCHAIN_FILE)
    set(LITERTLM_TOOLCHAIN_FILE "-DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE}" CACHE PATH "")
endif()

if(DEFINED LITERTLM_TOOLCHAIN_SCRIPT)
    message(STATUS "[LiteRTLM] Executing toolchain script -> ${LITERTLM_TOOLCHAIN_SCRIPT}")
    include("${LITERTLM_TOOLCHAIN_SCRIPT}")
endif()

set(LITERTLM_FBS_DIR "${LITERTLM_PROJECT_ROOT}/schema/core")
set(LITERTLM_FLATBUFFERS_FILES "litertlm_header_schema.fbs")

ExternalProject_Add(
  litert_lm
  DEPENDS      prebuild
  SOURCE_DIR   "${LITERTLM_PACKAGE_SOURCE_DIR}"
  PREFIX       "${LITERTLM_DIR}"
  BINARY_DIR   "${LITERTLM_BINARY_DIR}"
  STAMP_DIR    "${LITERTLM_STAMP_DIR}"
  TMP_DIR      "${LITERTLM_TMP_DIR}"
  DOWNLOAD_DIR "${LITERTLM_DOWNLOAD_DIR}"
  INSTALL_DIR  "${LITERTLM_INSTALL_DIR}"
  PATCH_COMMAND
      ${CMAKE_COMMAND} -E echo "[LiteRTLM] Generating FlatBuffers headers for LiteRT-LM..."
      COMMAND ${LITERTLM_HOST_FLATC} --cpp
      --gen-object-api --reflect-names
      --gen-mutable -o "${LITERTLM_FBS_DIR}"
      "${LITERTLM_FBS_DIR}/${LITERTLM_FLATBUFFERS_FILES}"
  CONFIGURE_COMMAND
    ${CMAKE_COMMAND} -E env CC=${LITERTLM_TARGET_CC_COMPILER} CXX=${LITERTLM_TARGET_CXX_COMPILER}
    ${CMAKE_COMMAND} -S ${LITERTLM_PACKAGE_SOURCE_DIR} -B ${LITERTLM_BINARY_DIR}
    "-DCMAKE_INSTALL_PREFIX=${LITERTLM_INSTALL_DIR}"
    "-DLITERTLM_CCRS_CXXFLAGS_KEY=${LITERTLM_CCRS_CXXFLAGS_KEY}"
    "-DLITERTLM_CCRS_CFLAGS_KEY=${LITERTLM_CCRS_CFLAGS_KEY}"
    "-DLITERTLM_CCRS_CXXFLAGS_VAL=${LITERTLM_CCRS_CXXFLAGS_VAL}"
    "-DLITERTLM_CCRS_CFLAGS_VAL=${LITERTLM_CCRS_CFLAGS_VAL}"
    ${LITERTLM_TOOLCHAIN_ARGS}
    ${LITERTLM_TOOLCHAIN_FILE}
    "-DLITERTLM_TARGET_CC_COMPILER=${LITERTLM_TARGET_CC_COMPILER}"
    "-DLITERTLM_TARGET_CXX_COMPILER=${LITERTLM_TARGET_CXX_COMPILER}"
    "-DLITERTLM_TOOLCHAIN_FILE=${LITERTLM_TOOLCHAIN_FILE}"
    "-DLITERTLM_TOOLCHAIN_ARGS=${LITERTLM_TOOLCHAIN_ARGS}"
    "-DLITERTLM_ORCHESTRATION_PHASE=litert_lm"
    "-DLITERTLM_TARGET_PHASE=litert_lm"
    "-DLITERTLM_PROJECT_ROOT=${LITERTLM_PROJECT_ROOT}"
    "-DLITERTLM_HOST_PROTOC=${LITERTLM_HOST_PROTOC}"
    "-DLITERTLM_HOST_PROTOC_BIN_DIR=${LITERTLM_HOST_PROTOC_BIN_DIR}"
    "-DLITERTLM_HOST_FLATC=${LITERTLM_HOST_FLATC}"
    "-DLITERTLM_HOST_FLATC_BIN_DIR=${LITERTLM_HOST_FLATC_BIN_DIR}"
    "-DLITERTLM_PREBUILD_TARGETS=${LITERTLM_PREBUILD_TARGETS}"
    "-DLITERTLM_PREBUILD_EXTERNAL_PROJECT_DIR=${LITERTLM_PREBUILD_EXTERNAL_PROJECT_DIR}"
    "-DLITERTLM_BUILD_FETCH_CONTENT=ON"
    "-DLITERTLM_BUILD_EXTERNAL_PROJECTS=ON"
    "-DLITERTLM_BUILD_TARGETS=ON"
    "-DCMAKE_CXX_FLAGS=-w"
  INSTALL_COMMAND ""
  BUILD_ALWAYS TRUE
)
